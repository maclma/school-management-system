  GET    /api/timetable/course/:course_id  → Filter by course
  GET    /api/timetable/teacher/:teacher_id → Filter by teacher
  GET    /api/timetable/day/:day           → Filter by day
  POST   /api/timetable                    → Create new
  PUT    /api/timetable/:id                → Update
  DELETE /api/timetable/:id                → Delete

  GET    /api/student/enrollments          → My enrollments
  GET    /api/enrollments/by-student/:id   → Student enrollments
  GET    /api/enrollments/by-course/:id    → Course enrollments
  PUT    /api/enrollments/:id/status       → Update status
